### Step 2.3 — Fonction create_index_if_absent

def create_index_if_absent(pc, index_name, dim=384, metric='cosine'):
    """Creer index Pinecone si absent. Confirmation utilisateur avant creation API.
    
    Args:
        pc: Pinecone client
        index_name: nom index <project-name>-memory
        dim: 384 (sentence-transformers/all-MiniLM-L6-v2)
        metric: cosine
    
    Returns:
        True si cree ou existe deja, False si refuse par utilisateur
    """
    existing = [idx['name'] for idx in pc.list_indexes()]
    if index_name in existing:
        print(f"[OK] Index '{index_name}' existe deja")
        return True
    
    # Garde-fou regle 13 : confirmation utilisateur
    confirm = input(f"[CONFIRM] Creer index Pinecone '{index_name}' ? (y/N): ")
    if confirm.lower() != 'y':
        print("[SKIP] Creation annulee par utilisateur")
        return False
    
    pc.create_index(
        name=index_name,
        dimension=dim,
        metric=metric,
        spec=ServerlessSpec(cloud='aws', region='us-east-1')
    )
    print(f"[NEW] Index '{index_name}' cree (dim={dim}, metric={metric})")
    return True
